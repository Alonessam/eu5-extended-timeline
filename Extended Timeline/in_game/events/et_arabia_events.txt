# .51 - Himyar and the People of the Book (records: arabia_himyar_book_religions,
# arabia_aksumite_yemen_525)
et_culture.51 = {
	type = country_event
	title = et_culture.51.title
	desc = et_culture.51.desc
	outcome = neutral
	historical_info = et_culture.51.historical_info

	illustration_tags = {
		10 = regular
		10 = interior
	}

	trigger = {
		current_year >= 300
		current_year <= 530
		OR = {
			owns = location:sana_yemen
			owns = location:marib
			owns = location:najran
		}
		NOT = { has_variable = et_arabia_2_fired }
	}

	immediate = {
		set_variable = { name = et_arabia_2_fired value = 1 }
	}

	option = {
		name = et_culture.51.a
		historical_option = yes
		set_variable = { name = et_arabia_book_religions value = 1 }
		if = {
			limit = {
				OR = {
					religion = religion:south_arabian_polytheism
					religion = religion:nabataean_religion
				}
			}
			change_religion = religion:judaism
			change_religion_for_ruler_and_family = { country = ROOT religion = religion:judaism }
		}
		custom_tooltip = {
			text = et_culture.51.a.tt
			every_owned_location = {
				limit = { area = area:north_yemen_area }
				every_pop = {
					limit = { religion = religion:south_arabian_polytheism }
					split_pop = { fraction = 0.20 religion = religion:judaism }
				}
			}
			every_owned_location = {
				limit = { area = area:south_yemen_area }
				every_pop = {
					limit = { religion = religion:south_arabian_polytheism }
					split_pop = { fraction = 0.05 religion = religion:miaphysite }
				}
			}
		}
		add_prestige = prestige_mild_bonus
		add_stability = -5
	}

	option = {
		name = et_culture.51.b
		set_variable = { name = et_arabia_2_resist value = 1 }
		custom_tooltip = {
			text = et_culture.51.b.tt
			every_owned_location = {
				limit = { area = area:north_yemen_area }
				every_pop = {
					limit = { religion = religion:south_arabian_polytheism }
					split_pop = { fraction = 0.05 religion = religion:judaism }
				}
			}
			every_owned_location = {
				limit = { area = area:south_yemen_area }
				every_pop = {
					limit = { religion = religion:south_arabian_polytheism }
					split_pop = { fraction = 0.02 religion = religion:miaphysite }
				}
			}
		}
		add_stability = 10
	}
}
