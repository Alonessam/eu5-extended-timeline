# 1. Concilium Galliarum & Claudian Integration (MS 58–100)
et_culture.1 = {
	type = country_event
	title = et_culture.1.title
	desc = et_culture.1.desc
	outcome = good
	historical_info = et_culture.1.historical_info

	fire_only_once = yes
	dynamic_historical_event = {
		tag = ROM
		from = 58.1.1
		to = 100.1.1
		monthly_chance = 15
	}

	trigger = {
		has_or_had_tag = ROM
		owns = location:lyon
	}

	option = {
		name = et_culture.1.a
		historical_option = yes
		custom_tooltip = {
			text = et_culture.1.tt
			every_owned_location = {
				limit = {
					OR = {
						location = location:lyon
						location = location:paris
						location = location:bordeaux
						location = location:toulouse
						location = location:nimes
						location = location:arles
						location = location:marseille
						location = location:reims
						location = location:rouen
					}
				}
				every_pop = {
					limit = {
						culture = culture:gallian
						OR = {
							pop_type = pop_type:nobles
							pop_type = pop_type:clergy
							pop_type = pop_type:burghers
						}
					}
					change_pop_culture = culture:gallo_roman
				}
			}
		}
		add_prestige = prestige_mild_bonus
	}

	option = {
		name = et_culture.1.b
		add_stability = -10
		add_prestige = -prestige_mild_bonus
	}
}
