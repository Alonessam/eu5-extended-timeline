et_culture.1 = { # Christianization I (record: egypt_christianization_rise)
	type = country_event
	title = et_culture.1.title
	desc = et_culture.1.desc
	outcome = neutral
	historical_info = et_culture.1.historical_info

	illustration_tags = {
		0 = regular
		1 = interior
	}

	trigger = {
		current_year >= 200
		current_year <= 500
		NOT = { has_variable = et_egypt_1_fired }
	}

	immediate = {
		# One-shot marker set at fire time, so the pulse cannot open a second
		# popup while the choice is still pending.
		set_variable = { name = et_egypt_1_fired value = 1 }
	}

	option = {
		name = et_culture.1.a
		historical_option = yes
		set_variable = { name = et_egypt_christ_1 value = 1 }
		custom_tooltip = {
			text = et_culture.1.a.tt
			# Lower Egypt (Delta): the towns lead the conversion.
			every_owned_location = {
				limit = {
					area = area:lower_egypt_area
				}
				every_pop = {
					limit = {
						culture = culture:old_egyptian
						religion = religion:egyptian_polytheism
					}
					split_pop = {
						fraction = 0.30
						religion = religion:miaphysite
					}
				}
			}
			# Upper Egypt: the conversion arrives later and slower.
			every_owned_location = {
				limit = {
					area = area:upper_egypt_area
				}
				every_pop = {
					limit = {
						culture = culture:old_egyptian
						religion = religion:egyptian_polytheism
					}
					split_pop = {
						fraction = 0.15
						religion = religion:miaphysite
					}
				}
			}
		}
	}

	option = {
		name = et_culture.1.b
		set_variable = { name = et_egypt_christ_resist value = 1 }
		custom_tooltip = {
			text = et_culture.1.b.tt
			# Resistance: a trickle only, and the chain stalls here.
			every_owned_location = {

					area = area:lower_egypt_area
